procedure NumStr (n : longint; var s : Str255);
{ From LMX-MAIN. }
var tmp : string[1];
    d   : integer;
    neg : boolean;
begin
   neg := n < 0;
   if neg then n := -n;
   if n = 0 then
      s := '0'
   else begin
      s := '';
      while n > 0 do begin
         d := n mod 10;
         tmp := '0';
         tmp[1] := chr(ord('0') + d);
         s := concat(tmp, s);
         n := n div 10
      end
   end;
   if neg then s := concat('-', s)
end;

function NextRand (n : integer) : integer;
{ 0 .. n-1.  A small multiplicative generator (no longint overflow:    }
{ seed stays below 65537).  Stirred with the idle counter so Lisa does }
{ not play the same game every time.                                   }
begin
   seed := (seed * 75 + 74 + (idleCnt mod 997)) mod 65537;
   if seed < 0 then seed := -seed;
   NextRand := (seed div 7) mod n
end;

{$R-}
{ Range checking off for the search: every index below is inside the   }
{ 10 x 10 board by construction (the border of cEdge cells stops each   }
{ walk), and this is where the time goes.  Back on after ChooseMove.    }

function CountFlips (var b : TBoard; cell, me : integer; doIt : boolean;
                     var wsum : integer) : integer;
{ How many discs a move by 'me' at 'cell' would turn (0 = not a legal   }
{ move), and in wsum the total weight of those discs (REV C, for the    }
{ carried score).  With doIt the move is made on b.  The caller has     }
{ checked that the cell is empty.                                        }
var d, dd, p, k, n, w, opp : integer;
begin
   n := 0;
   wsum := 0;
   opp := 3 - me;
   for d := 1 to 8 do begin
      dd := dirs[d];
      p := cell + dd;
      k := 0;
      w := 0;
      while b[p] = opp do begin
         w := w + weight[p];
         p := p + dd;
         k := k + 1
      end;
      if k > 0 then
         if b[p] = me then begin
            n := n + k;
            wsum := wsum + w;
            if doIt then begin
               p := cell + dd;
               while b[p] = opp do begin
                  b[p] := me;
                  p := p + dd
               end
            end
         end
   end;
   if doIt then
      if n > 0 then b[cell] := me;
   CountFlips := n
end;

function Near (var b : TBoard; cell, opp : integer) : boolean;
{ TRUE if one of the eight neighbours holds an opponent disc -- the only }
{ empties that can be legal moves.                                       }
begin
   Near := (b[cell - 11] = opp) or (b[cell - 10] = opp) or (b[cell - 9] = opp)
        or (b[cell - 1] = opp)  or (b[cell + 1] = opp)
        or (b[cell + 9] = opp)  or (b[cell + 10] = opp) or (b[cell + 11] = opp)
end;

procedure GenMoves (var b : TBoard; me : integer; var ml : TMoveList; var n : integer);
{ The legal moves for 'me', best cells first (the 'order' list). }
var i, cell, opp, w : integer;
begin
   n := 0;
   opp := 3 - me;
   for i := 1 to 64 do begin
      cell := order[i];
      if b[cell] = cEmpty then
         if Near(b, cell, opp) then
            if CountFlips(b, cell, me, false, w) > 0 then begin
               n := n + 1;
               ml[n] := cell
            end
   end
end;

function HasMove (var b : TBoard; me : integer) : boolean;
var i, cell, opp, w : integer;
    found           : boolean;
begin
   found := false;
   opp := 3 - me;
   i := 1;
   while (i <= 64) and (not found) do begin
      cell := order[i];
      if b[cell] = cEmpty then
         if Near(b, cell, opp) then
            if CountFlips(b, cell, me, false, w) > 0 then
               found := true;
      i := i + 1
   end;
   HasMove := found
end;

procedure CountDiscs (var b : TBoard; var nBlack, nWhite : integer);
var r, c, v : integer;
begin
   nBlack := 0;
   nWhite := 0;
   for r := 1 to 8 do
      for c := 1 to 8 do begin
         v := b[r * 10 + c];
         if v = cBlack then nBlack := nBlack + 1
         else if v = cWhite then nWhite := nWhite + 1
      end
end;

{------------------------------------------------------------------------}
{ Lisa's search (REV C): negamax with alpha-beta, the score carried down }
{ the recursion.  pos = positional score and dd = disc difference, both  }
{ from the side to move's point of view; emp = empty cells.  A move at   }
{ cell c turning k discs of total weight w changes them by               }
{ weight[c] + 2w and 1 + 2k.  The leaf value is pos until the last 12    }
{ empties, then dd x 10; the end of the game (no empties, or neither side }
{ can move) is dd x 100 so a won game outranks any positional score.    }
{------------------------------------------------------------------------}

function LeafVal (pos, dd, emp : integer) : integer;
begin
   if emp <= 0 then
      LeafVal := dd * 100
   else if emp <= 12 then
      LeafVal := dd * 10
   else
      LeafVal := pos
end;

function Search (var b : TBoard; me, depth, alpha, beta, pos, dd, emp : integer) : integer;
{ depth 0 = the leaf value.  depth 1 = the last ply folded in: each     }
{ legal move is valued from its flips alone, no board copy.  A side    }
{ with no move passes (depth still counts down); no move for either    }
{ side ends the game.                                                   }
var nb   : TBoard;
    ml   : TMoveList;
    nm   : integer;
    i    : integer;
    sc   : integer;
    best : integer;
    k, w : integer;
    cell : integer;
    opp  : integer;
    cut  : boolean;
begin
   nodes := nodes + 1;
   opp := 3 - me;
   if depth <= 0 then begin
      Search := LeafVal(pos, dd, emp);
      exit(Search)
   end;
   if depth = 1 then begin
      { REV D: stop at the first candidate that reaches beta (the      }
      { alpha-beta cutoff the interior loop has always had).            }
      best := -30000;
      nm := 0;
      i := 1;
      cut := false;
      while (i <= 64) and (not cut) do begin
         cell := order[i];
         if b[cell] = cEmpty then
            if Near(b, cell, opp) then begin
               k := CountFlips(b, cell, me, false, w);
               if k > 0 then begin
                  nm := nm + 1;
                  nodes := nodes + 1;
                  sc := LeafVal(pos + weight[cell] + 2 * w, dd + 1 + 2 * k, emp - 1);
                  if sc > best then best := sc;
                  if best >= beta then cut := true
               end
            end;
         i := i + 1
      end;
      if nm > 0 then
         Search := best
      else if HasMove(b, opp) then
         Search := -LeafVal(-pos, -dd, emp)
      else
         Search := dd * 100;
      exit(Search)
   end;
   GenMoves(b, me, ml, nm);
   if nm = 0 then begin
      if HasMove(b, opp) then
         Search := -Search(b, opp, depth - 1, -beta, -alpha, -pos, -dd, emp)
      else
         Search := dd * 100;
      exit(Search)
   end;
   best := -30000;
   cut := false;
   i := 1;
   while (i <= nm) and (not cut) do begin
      nb := b;
      cell := ml[i];
      k := CountFlips(nb, cell, me, true, w);
      sc := -Search(nb, opp, depth - 1, -beta, -alpha,
                    -(pos + weight[cell] + 2 * w), -(dd + 1 + 2 * k), emp - 1);
      if sc > best then best := sc;
      if best > alpha then alpha := best;
      if alpha >= beta then cut := true;
      i := i + 1
   end;
   Search := best
end;

function ChooseMove : integer;
{ Lisa's move at the current level: the best root move, ties broken at  }
{ random.  The root window is (-30000, 1 - best) so a move that equals  }
{ the best so far is still scored exactly and can take part in a tie.   }
{ REV C: the root's pos / dd / emp are summed once here; from           }
{ solveAt[level] empties the depth reaches the end of the game.         }
var ml    : TMoveList;
    nm    : integer;
    i     : integer;
    sc    : integer;
    best  : integer;
    nTies : integer;
    pick  : integer;
    nb    : TBoard;
    depth : integer;
    k, w  : integer;
    cell  : integer;
    pos   : integer;
    dd    : integer;
    emp   : integer;
    r, c  : integer;
    v     : integer;
begin
   nodes := 0;
   pos := 0;
   dd := 0;
   emp := 0;
   for r := 1 to 8 do
      for c := 1 to 8 do begin
         cell := r * 10 + c;
         v := board[cell];
         if v = cEmpty then
            emp := emp + 1
         else if v = lisaC then begin
            pos := pos + weight[cell];
            dd := dd + 1
         end
         else begin
            pos := pos - weight[cell];
            dd := dd - 1
         end
      end;
   depth := level;
   if solveAt[level] > 0 then
      if emp <= solveAt[level] then depth := emp + 4;
   GenMoves(board, lisaC, ml, nm);
   pick := 0;
   best := -20000;
   nTies := 0;
   for i := 1 to nm do begin
      nb := board;
      cell := ml[i];
      k := CountFlips(nb, cell, lisaC, true, w);
      sc := -Search(nb, human, depth - 1, -30000, 1 - best,
                    -(pos + weight[cell] + 2 * w), -(dd + 1 + 2 * k), emp - 1);
      if sc > best then begin
         best := sc;
         pick := cell;
         nTies := 1
      end
      else if sc = best then begin
         nTies := nTies + 1;
         if NextRand(nTies) = 0 then pick := cell
      end
   end;
   ChooseMove := pick
end;
{$R+}

procedure DrawCell (cell : integer);
{ One cell: the disc, or a legal-move dot; Lisa's last-move mark in the }
{ top-left corner; the cursor as a double frame.                        }
var r, c, i, v : integer;
    rc, d      : Rect;
    isLegal    : boolean;
begin
   r := cell div 10;
   c := cell mod 10;
   CellRect(r, c, rc);
   SetRect(d, rc.left + 1, rc.top + 1, rc.right, rc.bottom);
   EraseRect(d);
   v := board[cell];
   SetRect(d, rc.left + 3, rc.top + 3, rc.right - 3, rc.bottom - 3);
   if v = cBlack then
      PaintOval(d)
   else if v = cWhite then
      FrameOval(d)
   else if showDots and (not thinking) and (not gameOver) then begin
      isLegal := false;
      for i := 1 to nLegal do
         if legal[i] = cell then isLegal := true;
      if isLegal then begin
         SetRect(d, rc.left + 14, rc.top + 14, rc.left + 20, rc.top + 20);
         PaintOval(d)
      end
   end;
   if cell = lastLisa then begin
      SetRect(d, rc.left + 2, rc.top + 2, rc.left + 6, rc.top + 6);
      PaintRect(d)
   end;
   if (r = cursorR) and (c = cursorC) then begin
      SetRect(d, rc.left + 1, rc.top + 1, rc.right, rc.bottom);
      FrameRect(d);
      SetRect(d, rc.left + 2, rc.top + 2, rc.right - 1, rc.bottom - 1);
      FrameRect(d)
   end
end;

      x := boardX + i * cellPx;

      DrawString(s)

      for c := 1 to 8 do
         DrawCell(r * 10 + c)

procedure StartTurn;
{ Your turn: find your legal moves, park the cursor on one if it is not }
{ already, and repaint.                                                  }
var i  : integer;
    on : boolean;
begin
   GenMoves(board, human, legal, nLegal);
   on := false;
   for i := 1 to nLegal do
      if legal[i] = cursorR * 10 + cursorC then on := true;
   if (not on) and (nLegal > 0) then begin
      cursorR := legal[1] div 10;
      cursorC := legal[1] mod 10
   end;
   DrawShow
end;

procedure FinishGame;
{ Neither side can move (or the board is full). }
var nb, nw     : integer;
    mine, hers : integer;
    s, t       : Str255;
begin
   gameOver := true;
   thinking := false;
   nLegal := 0;
   CountDiscs(board, nb, nw);
   if human = cBlack then begin
      NumStr(nb, s);                  { s = yours, t = Lisa's }
      NumStr(nw, t);
      mine := nb;
      hers := nw
   end
   else begin
      NumStr(nw, s);
      NumStr(nb, t);
      mine := nw;
      hers := nb
   end;
   if mine > hers then
      status := concat('Game over: you win, ', concat(s, concat(' to ', t)))
   else if hers > mine then
      status := concat('Game over: Lisa wins, ', concat(t, concat(' to ', s)))
   else
      status := concat('Game over: a draw, ', concat(s, concat(' to ', t)));
   DrawShow;
   if mine > hers then
      NoteAlert(allAlerts, alrtWin)
   else if hers > mine then
      NoteAlert(allAlerts, alrtLose)
   else
      NoteAlert(allAlerts, alrtDraw)
end;

procedure LisaTurn;
{ Lisa moves; again if you then have no reply.  Ends with your turn or  }
{ the end of the game.                                                   }
var mv, n, w : integer;
    again    : boolean;
begin
   repeat
      again := false;
      if HasMove(board, lisaC) then begin
         thinking := true;
         status := 'Lisa is thinking...';
         DrawPanel;
         mv := ChooseMove;
         n := CountFlips(board, mv, lisaC, true, w);
         lastLisa := mv;
         thinking := false;
         if HasMove(board, human) then
            status := 'Your move.'
         else if HasMove(board, lisaC) then begin
            status := 'You have no move -- Lisa moves again.';
            DrawShow;
            NoteAlert(allAlerts, alrtUSkip);
            again := true
         end
         else begin
            FinishGame;
            exit(LisaTurn)
         end
      end
      else begin
         if HasMove(board, human) then begin
            status := 'Lisa has no move -- your turn again.';
            DrawShow;
            NoteAlert(allAlerts, alrtLSkip)
         end
         else begin
            FinishGame;
            exit(LisaTurn)
         end
      end
   until not again;
   StartTurn
end;

procedure PlayerMove (cell : integer);
{ Your move at cell, if it is legal; then Lisa's reply. }
var n, w : integer;
    s    : Str255;
begin
   if (myWindow = nil) or gameOver or thinking then
      exit(PlayerMove);
   if board[cell] <> cEmpty then
      n := 0
   else
      n := CountFlips(board, cell, human, false, w);
   if n = 0 then begin
      CellName(cell, s);
      status := concat('No flips at ', concat(s, ' -- try another cell.'));
      SyncPort;
      DrawPanel;
      exit(PlayerMove)
   end;
   if nHist < maxHist then begin
      nHist := nHist + 1;
      hist[nHist] := board
   end;
   n := CountFlips(board, cell, human, true, w);
   lastYou := cell;
   started := true;
   cursorR := cell div 10;
   cursorC := cell mod 10;
   nLegal := 0;
   thinking := true;
   status := 'Lisa is thinking...';
   DrawShow;
   LisaTurn
end;

procedure NewGame;
var i, r, c : integer;
begin
   for i := 0 to 99 do board[i] := cEdge;
   for r := 1 to 8 do
      for c := 1 to 8 do board[r * 10 + c] := cEmpty;
   board[44] := cWhite;               { d4 }
   board[45] := cBlack;               { e4 }
   board[54] := cBlack;               { d5 }
   board[55] := cWhite;               { e5 }
   nHist := 0;
   gameOver := false;
   started := false;
   thinking := false;
   lastYou := 0;
   lastLisa := 0;
   nodes := 0;
   cursorR := 3;                      { d3, a legal opening move }
   cursorC := 4;
   if human = cBlack then begin
      status := 'Your move.  Black plays first.';
      StartTurn
   end
   else begin
      status := 'Lisa is black and opens.';
      DrawShow;
      LisaTurn
   end
end;

procedure UndoMove;
{ Back to the position before your last move (Lisa's reply goes too). }
begin
   if (myWindow = nil) or thinking then
      exit(UndoMove);
   if nHist = 0 then begin
      status := 'Nothing to take back.';
      SyncPort;
      DrawPanel;
      exit(UndoMove)
   end;
   board := hist[nHist];
   nHist := nHist - 1;
   gameOver := false;
   lastYou := 0;
   lastLisa := 0;
   nodes := 0;                        { REV C: the count belonged to the undone move }
   started := nHist > 0;
   status := 'Took back your last move and the reply.';
   StartTurn
end;

procedure AskNewGame;
begin
   if started and (not gameOver) then begin
      if CautionAlert(allAlerts, alrtNewG) then NewGame
   end
   else
      NewGame
end;

procedure SwitchSides;
{ REV C: a new game with the colours swapped; Lisa opens when she is    }
{ black.  Same caution as New Game when a game is in progress.          }
begin
   if started and (not gameOver) then
      if not CautionAlert(allAlerts, alrtNewG) then
         exit(SwitchSides);
   human := 3 - human;
   lisaC := 3 - human;
   NewGame
end;

procedure SetLevel (n : integer);
begin
   if (n >= 1) and (n <= maxLevel) then begin
      level := n;
      SyncPort;
      DrawPanel
   end
end;

procedure ToggleDots;
begin
   showDots := not showDots;
   DrawShow
end;

procedure MoveCursor (dr, dc : integer);
var old : integer;
begin
   if myWindow = nil then
      exit(MoveCursor);
   old := cursorR * 10 + cursorC;
   cursorR := cursorR + dr;
   cursorC := cursorC + dc;
   if cursorR < 1 then cursorR := 1;
   if cursorR > 8 then cursorR := 8;
   if cursorC < 1 then cursorC := 1;
   if cursorC > 8 then cursorC := 8;
   SyncPort;
   DrawCell(old);
   DrawCell(cursorR * 10 + cursorC);
   DrawPanel
end;

procedure ClickCell (where : Point);
{ A click on the board plays that cell.  where is window-local. }
var r, c : integer;
begin
   if myWindow = nil then
      exit(ClickCell);
   if (where.h < boardX) or (where.v < boardY) then
      exit(ClickCell);
   if (where.h >= boardX + boardPx) or (where.v >= boardY + boardPx) then
      exit(ClickCell);
   c := (where.h - boardX) div cellPx + 1;
   r := (where.v - boardY) div cellPx + 1;
   cursorR := r;
   cursorC := c;
   PlayerMove(r * 10 + c)
end;

procedure MenuCommand (menu, item : integer);
var err : integer;
begin
   case menu of

      mFile :
         case item of
            miAsideAll : DoFilingCmd(cmdClosAll);
            miAside    : DoFilingCmd(cmdClose);
            miPutAway  :
               begin
                  TellFiler(err, docClosd, docPutBack, myWindow);
                  myWindow := nil
               end;
         end;

      mGame :
         case item of
            miNew  : AskNewGame;
            miUndo : UndoMove;
            miDots  : ToggleDots;
            miSides : SwitchSides;
         end;

      mLevel :
         SetLevel(item);

      mHelp :
         case item of
            miHowTo : NoteAlert(allAlerts, alrtHow);
            miKeys  : NoteAlert(allAlerts, alrtKeys);
         end;

      mVer :
         case item of
            miAbout : NoteAlert(allAlerts, alrtAbout);
         end;

   end
end;

procedure PlainKey (ch : char);
{ Keys without the Apple key: keypad 4 6 8 2 (or J L I K) move the      }
{ cursor, RETURN / SPACE play it, N new, U undo, D dots, 1-4 level,     }
{ H help.                                                                }
begin
   if (ch = '4') or (ch = 'J') or (ch = 'j') then
      MoveCursor(0, -1)
   else if (ch = '6') or (ch = 'L') or (ch = 'l') then
      MoveCursor(0, 1)
   else if (ch = '8') or (ch = 'I') or (ch = 'i') then
      MoveCursor(-1, 0)
   else if (ch = '2') or (ch = 'K') or (ch = 'k') then
      MoveCursor(1, 0)
   else if (ch = chr(13)) or (ch = chr(3)) or (ch = ' ') then
      PlayerMove(cursorR * 10 + cursorC)
   else if (ch = 'N') or (ch = 'n') then
      AskNewGame
   else if (ch = 'U') or (ch = 'u') then
      UndoMove
   else if (ch = 'D') or (ch = 'd') then
      ToggleDots
   else if (ch = 'S') or (ch = 's') then
      SwitchSides
   else if (ch >= '1') and (ch <= '4') then
      SetLevel(ord(ch) - ord('0'))
   else if (ch = 'H') or (ch = 'h') or (ch = '?') then
      NoteAlert(allAlerts, alrtHow)
end;

procedure ProcessTheEvent;
var theMenu, theItem : integer;
begin
   case theEvent.what of

      buttonDown :
         if theEvent.who = menuFolder then begin
            MenuSelect(theEvent.where, theMenu, theItem);
            if (theMenu <> 0) and (theItem <> 0) then
               MenuCommand(theMenu, theItem);
            HiLiteMenu(0)
         end
         else if theEvent.who = myWindow then begin
            seed := seed + theEvent.where.h * 7 + theEvent.where.v * 13;
            ClickCell(theEvent.where)
         end;

      buttonUp :
         ;

      keyDown :
         if (theEvent.who = myWindow) and (not theEvent.repeatKey) then begin
            seed := seed + ord(theEvent.ascii) + 31;
            if theEvent.appleKey then begin
               MenuKey(theEvent.ascii, theMenu, theItem);
               if theMenu <> 0 then begin
                  HiLiteMenu(theMenu);
                  MenuCommand(theMenu, theItem);
                  HiLiteMenu(0)
               end
            end
            else
               PlainKey(theEvent.ascii)
         end;

      folderActivate :
         begin
            TakeControl(theEvent, false, false);
            PutUpMenus
         end;

      folderDeactivate :
         GiveControl(theEvent);

      folderUpdate :
         if theEvent.who = myWindow then
            doUpdateEvent(true, theEvent.who)
         else begin
            BeginUpdate(theEvent.who);
            EndUpdate(theEvent.who)
         end;

      folderMoved :
         if theEvent.who = myWindow then begin
            SyncPort;
            DrawShow
         end;

      filerEvent :
         DoFilerEvent;

      abortEvent, diedEvent, diskEvent, catalogEvent, nilEvent :
         ;

      otherwise

   end
end;

{------------------------------------------------------------------------}
{ Initialize (from PZL-MAIN).  Also the weight table, the move order    }
{ and the direction steps.                                               }
{------------------------------------------------------------------------}

procedure Initialize;
var err      : integer;
    schedErr : integer;
    pcInfo   : procinfoRec;
    excName  : t_ex_name;
    myPrefix : TParamAlert;
    cutAt    : integer;
    i, r, c  : integer;
    k, w     : integer;
    n        : integer;
    ew       : array [1..4] of integer;
    vals     : array [1..7] of integer;
begin
   initing := true;

   excName := 'SYS_TERMINATE';
   Declare_Excep_Hdl(err, excName, @LastWishes);
   if err <> 0 then begin
      TellFiler(err, initFailed, noMemory, nil);
      halt
   end;

   toldToDie := false;
   myWindow  := nil;
   human     := cBlack;
   lisaC     := cWhite;
   solveAt[1] := 0;                   { REV C: exact endgame from this }
   solveAt[2] := 0;                   { many empties, per level }
   solveAt[3] := 8;
   solveAt[4] := 9;
   level     := 2;
   showDots  := true;
   gameOver  := false;
   started   := false;
   thinking  := false;
   nHist     := 0;
   nLegal    := 0;
   cursorR   := 3;
   cursorC   := 4;
   lastYou   := 0;
   lastLisa  := 0;
   nodes     := 0;
   status    := '';
   seed      := 12345;
   idleCnt   := 0;
   for i := 0 to 99 do board[i] := cEdge;

   { the eight direction steps on the 10 x 10 board }
   dirs[1] := -11;  dirs[2] := -10;  dirs[3] := -9;  dirs[4] := -1;
   dirs[5] := 1;    dirs[6] := 9;    dirs[7] := 10;  dirs[8] := 11;

   { positional weights: edge cells by distance from the corner (1 = the }
   { corner, 2 = the C square, 3, 4), X squares -50, the ring inside the }
   { edge -2, the middle -1.                                              }
   ew[1] := 100;  ew[2] := -20;  ew[3] := 10;  ew[4] := 5;
   for i := 0 to 99 do weight[i] := 0;
   for r := 1 to 8 do
      for c := 1 to 8 do begin
         w := -1;
         if (r = 2) or (r = 7) or (c = 2) or (c = 7) then w := -2;
         if ((r = 2) or (r = 7)) and ((c = 2) or (c = 7)) then w := -50;
         if (r = 1) or (r = 8) or (c = 1) or (c = 8) then begin
            { an edge cell: k = distance along the edge from the nearest }
            { corner (1 = the corner itself, 2 = the C square ...)       }
            if (r = 1) or (r = 8) then k := c else k := r;
            if k > 4 then k := 9 - k;
            w := ew[k]
         end;
         weight[r * 10 + c] := w
      end;

   { the move order: best weights first }
   vals[1] := 100;  vals[2] := 10;  vals[3] := 5;  vals[4] := -1;
   vals[5] := -2;   vals[6] := -20; vals[7] := -50;
   n := 0;
   for i := 1 to 7 do
      for r := 1 to 8 do
         for c := 1 to 8 do
            if weight[r * 10 + c] = vals[i] then begin
               n := n + 1;
               order[n] := r * 10 + c
            end;

   InitHeap(@myZone[0], @myZone[maxHeap], @OutOfHeapSpace);

   Sched_Class(schedErr, false);
   Info_Process(err, My_ID, pcInfo);
   Sched_Class(schedErr, true);

   cutAt := pos('}', pcInfo.progpathname);
   if (cutAt = 0) or (cutAt > 40) then begin
      TellFiler(err, initFailed, internalError, nil);
      halt
   end;
   myPrefix := copy(pcInfo.progpathname, 1, cutAt);

   InitMP(theHeap, myPrefix, @AlertFailed, alertVerNum,
          @myMenus[mFile], allAlerts, err);
   if err > 0 then begin
      TellFiler(err, initFailed, noMemory, nil);
      halt
   end;

   InitWmlCrs(err);
   if err <> 0 then begin
      TellFiler(err, initFailed, noDiskSpace, nil);
      halt
   end;

   SetPort(deskPort);

   initing := false
end;

{------------------------------------------------------------------------}
{ The main program -- two-phase loop; the idle branch only stirs the     }
{ random seed.                                                           }
{------------------------------------------------------------------------}

begin { main program }

   InitGraf(@thePort);
   OpenWm;
   Initialize;

   repeat

      while not (ImActive or toldToDie) do begin
         GetEvent(theEvent);
         ProcessTheEvent
      end;

      while ImActive and (not toldToDie) do begin
         if EventAvail then begin
            GetEvent(theEvent);
            ProcessTheEvent
         end
         else begin
            idleCnt := idleCnt + 1;
            if idleCnt > 1000000 then idleCnt := 0;
            LetOthersRun
         end
      end;

   until toldToDie;

   termErr := 0
end.
